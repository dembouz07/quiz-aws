const G = {
  t1:{name:"",pts:0,correct:0,wrong:0},
  t2:{name:"",pts:0,correct:0,wrong:0},
  questions:[],
  idx:0,
  total:10,
  timePerQ:20,
  timeLeft:20,
  timer:null,
  buzzed:0,
  answered:false,
  totalCorrect:0,
  totalAnswered:0,
  lastCat:"all"
};

function shuffle(a){const b=[...a];for(let i=b.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[b[i],b[j]]=[b[j],b[i]]}return b;}
function showScreen(id){document.querySelectorAll(".screen").forEach(s=>s.classList.remove("active"));document.getElementById(id).classList.add("active");}
function $(id){return document.getElementById(id);}

function startGame(){
  G.t1.name=$("team1Name").value.trim()||"Equipe 1";
  G.t2.name=$("team2Name").value.trim()||"Equipe 2";
  G.t1.pts=G.t1.correct=G.t1.wrong=0;
  G.t2.pts=G.t2.correct=G.t2.wrong=0;
  G.total=parseInt($("numQuestions").value);
  G.timePerQ=parseInt($("timePerQ").value);
  G.totalCorrect=G.totalAnswered=0;
  G.lastCat=$("categoryFilter").value;
  let pool=G.lastCat==="all"?QUESTIONS:QUESTIONS.filter(q=>q.c===G.lastCat);
  if(pool.length<G.total){const extra=shuffle(QUESTIONS.filter(q=>q.c!==G.lastCat));pool=[...shuffle(pool),...extra];}
  G.questions=shuffle(pool).slice(0,G.total);
  G.idx=0;
  $("displayName1").textContent=G.t1.name;
  $("displayName2").textContent=G.t2.name;
  showScreen("game");
  loadQ();
}

function loadQ(){
  if(G.idx>=G.questions.length){endGame();return;}
  const q=G.questions[G.idx];
  G.buzzed=0;G.answered=false;
  $("categoryBadge").textContent=q.c;
  $("questionText").textContent=q.q;
  $("progressBar").style.width=((G.idx/G.total)*100)+"%";
  $("progressLabel").textContent="Question "+(G.idx+1)+" / "+G.total;
  const letters=["A","B","C","D"];
  const order=shuffle([0,1,2,3]);
  for(let i=0;i<4;i++){
    const btn=$("ans"+i);
    btn.dataset.orig=order[i];
    btn.querySelector(".answer-letter").textContent=letters[i];
    btn.querySelector(".answer-text").textContent=q.a[order[i]];
    btn.className="answer-btn";
    btn.disabled=false;
  }
  $("buzzerZone").className="buzzer-zone waiting";
  $("buzzerStatus").className="buzzer-status";
  $("buzzerStatus").textContent="Appuyez sur Q ou P pour buzzer !";
  $("buzzerActions").style.display="none";
  $("explanationBox").className="explanation-box";
  $("nextBtn").className="btn-next";
  $("scoreCard1").classList.remove("buzzed");
  $("scoreCard2").classList.remove("buzzed");
  startTimer();
}

function startTimer(){
  clearInterval(G.timer);
  G.timeLeft=G.timePerQ;
  updateTimer();
  G.timer=setInterval(()=>{G.timeLeft--;updateTimer();if(G.timeLeft<=0){clearInterval(G.timer);timeUp();}},1000);
}

function updateTimer(){
  const t=G.timeLeft,max=G.timePerQ;
  $("timerText").textContent=t;
  const r=30,circ=2*Math.PI*r;
  const prog=$("timerCircle");
  prog.style.strokeDasharray=circ;
  prog.style.strokeDashoffset=circ*(1-t/max);
  const ratio=t/max;
  prog.style.stroke=ratio>0.5?"#22c55e":ratio>0.25?"#f59e0b":"#ef4444";
}

function timeUp(){
  if(G.answered)return;
  G.answered=true;
  revealAnswer();
  $("buzzerZone").className="buzzer-zone";
  $("buzzerStatus").textContent="Temps ecoule ! Aucun point.";
  $("buzzerActions").style.display="none";
  $("explanationBox").className="explanation-box show";
  $("explanationText").textContent=G.questions[G.idx].e;
  $("nextBtn").className="btn-next show";
}

document.addEventListener("keydown",e=>{
  const k=e.key.toLowerCase();
  if(k==="q"&&!G.buzzed&&!G.answered)buzz(1);
  else if(k==="p"&&!G.buzzed&&!G.answered)buzz(2);
});

function buzz(team){
  G.buzzed=team;
  clearInterval(G.timer);
  const name=team===1?G.t1.name:G.t2.name;
  $("buzzerZone").className="buzzer-zone buzzed-t"+(team===1?"1":"2");
  $("buzzerStatus").className="buzzer-status active"+(team===2?"-t2":"");
  $("buzzerStatus").textContent=name+" a buzze !";
  $("buzzerActions").style.display="flex";
  $("scoreCard1").classList.remove("buzzed");
  $("scoreCard2").classList.remove("buzzed");
  $("scoreCard"+team).classList.add("buzzed");
  playBuzz(team);
}

function selectAnswer(btnIdx){
  if(G.answered)return;
  const orig=parseInt($("ans"+btnIdx).dataset.orig);
  const correct=orig===G.questions[G.idx].r;
  if(G.buzzed){judgeAnswer(correct,btnIdx);}
  else{
    G.buzzed=0;
    judgeAnswer(correct,btnIdx);
  }
}

function judgeAnswer(correct,btnIdx){
  if(G.answered)return;
  G.answered=true;
  clearInterval(G.timer);
  G.totalAnswered++;
  revealAnswer(btnIdx);
  if(correct){
    G.totalCorrect++;
    if(G.buzzed===1){G.t1.pts+=2;G.t1.correct++;showFeedback(true,G.t1.name);}
    else if(G.buzzed===2){G.t2.pts+=2;G.t2.correct++;showFeedback(true,G.t2.name);}
    else showFeedback(true,"");
    playCorrect();
  }else{
    if(G.buzzed===1){G.t1.pts=Math.max(0,G.t1.pts-1);G.t1.wrong++;showFeedback(false,G.t1.name);}
    else if(G.buzzed===2){G.t2.pts=Math.max(0,G.t2.pts-1);G.t2.wrong++;showFeedback(false,G.t2.name);}
    else showFeedback(false,"");
    playWrong();
  }
  $("score1").textContent=G.t1.pts;
  $("score2").textContent=G.t2.pts;
  $("scoreCard1").classList.remove("buzzed");
  $("scoreCard2").classList.remove("buzzed");
  $("buzzerZone").className="buzzer-zone";
  $("buzzerActions").style.display="none";
  $("explanationBox").className="explanation-box show";
  $("explanationText").textContent=G.questions[G.idx].e;
  $("nextBtn").className="btn-next show";
}

function revealAnswer(selectedBtnIdx){
  const correctOrig=G.questions[G.idx].r;
  for(let i=0;i<4;i++){
    const btn=$("ans"+i);
    btn.disabled=true;
    if(parseInt(btn.dataset.orig)===correctOrig)btn.classList.add("correct");
    else if(i===selectedBtnIdx)btn.classList.add("wrong");
  }
}

function showFeedback(correct,teamName){
  const ov=$("feedbackOverlay"),bb=$("feedbackBubble");
  $("fbIcon").textContent=correct?"✅":"❌";
  $("fbText").textContent=correct?(teamName?"+2 pts pour "+teamName+" !":"Bonne reponse !"):(teamName?"-1 pt pour "+teamName:"Mauvaise reponse !");
  bb.className="feedback-bubble "+(correct?"correct-fb":"wrong-fb");
  ov.className="feedback-overlay show";
  setTimeout(()=>{ov.className="feedback-overlay";},1500);
}

function nextQuestion(){G.idx++;loadQ();}

function endGame(){
  clearInterval(G.timer);
  $("resultName1").textContent=G.t1.name;
  $("resultName2").textContent=G.t2.name;
  $("resultScore1").textContent=G.t1.pts;
  $("resultScore2").textContent=G.t2.pts;
  $("statTotal").textContent=G.totalAnswered;
  $("statCorrect").textContent=G.totalCorrect;
  $("statAccuracy").textContent=G.totalAnswered>0?Math.round(G.totalCorrect/G.totalAnswered*100)+"%":"0%";
  if(G.t1.pts>G.t2.pts){
    $("winnerName").textContent=G.t1.name;
    $("winnerScore").textContent=G.t1.pts+" pts vs "+G.t2.pts+" pts";
    $("resultCard1").classList.add("winner-card");
    $("resultCard2").classList.remove("winner-card");
  }else if(G.t2.pts>G.t1.pts){
    $("winnerName").textContent=G.t2.name;
    $("winnerScore").textContent=G.t2.pts+" pts vs "+G.t1.pts+" pts";
    $("resultCard2").classList.add("winner-card");
    $("resultCard1").classList.remove("winner-card");
  }else{
    $("winnerName").textContent="Egalite !";
    $("winnerScore").textContent=G.t1.pts+" pts chacun";
    $("resultCard1").classList.remove("winner-card");
    $("resultCard2").classList.remove("winner-card");
  }
  showScreen("results");
}

function restartGame(){showScreen("home");}

function playBuzz(t){try{const c=new AudioContext(),o=c.createOscillator(),g=c.createGain();o.connect(g);g.connect(c.destination);o.frequency.value=t===1?440:330;o.type="square";g.gain.setValueAtTime(0.3,c.currentTime);g.gain.exponentialRampToValueAtTime(0.001,c.currentTime+0.3);o.start();o.stop(c.currentTime+0.3);}catch(e){}}
function playCorrect(){try{const c=new AudioContext();[523,659,784].forEach((f,i)=>{const o=c.createOscillator(),g=c.createGain();o.connect(g);g.connect(c.destination);o.frequency.value=f;o.type="sine";g.gain.setValueAtTime(0.2,c.currentTime+i*0.1);g.gain.exponentialRampToValueAtTime(0.001,c.currentTime+i*0.1+0.2);o.start(c.currentTime+i*0.1);o.stop(c.currentTime+i*0.1+0.2);});}catch(e){}}
function playWrong(){try{const c=new AudioContext(),o=c.createOscillator(),g=c.createGain();o.connect(g);g.connect(c.destination);o.frequency.value=200;o.type="sawtooth";g.gain.setValueAtTime(0.3,c.currentTime);g.gain.exponentialRampToValueAtTime(0.001,c.currentTime+0.5);o.start();o.stop(c.currentTime+0.5);}catch(e){}}
